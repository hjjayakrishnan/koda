

import math

def compute_reward(state: dict, action: dict) -> dict:
    """
    Reward components:
    - speed_huber: Huber loss for speed tracking (less sensitive to outliers).
    - yaw_huber: Huber loss for yaw rate tracking.
    - upright_cos: Cosine-based upright reward (max 1 at pitch=0).
    - joint_vel_penalty: Penalty on squared joint velocities.
    - torque_penalty: Penalty on squared joint torques.
    - contact_bonus: Small bonus for maintaining ground contact.
    """
    # Helper for Huber loss
    def huber(x, delta=1.0):
        abs_x = abs(x)
        if abs_x <= delta:
            return 0.5 * x * x
        else:
            return delta * (abs_x - 0.5 * delta)

    # Speed tracking
    cmd_speed = state.get('commanded_speed', 0.0)
    actual_speed = state.get('ground_velocity', 0.0)
    speed_err = cmd_speed - actual_speed
    speed_huber = -huber(speed_err, delta=0.5)  # negative because we want to minimise error

    # Yaw tracking
    cmd_yaw = state.get('commanded_yaw_rate', 0.0)
    actual_yaw = state.get('yaw_rate', 0.0)
    yaw_err = cmd_yaw - actual_yaw
    yaw_huber = -huber(yaw_err, delta=0.2)

    # Uprightness via cosine (cos(0)=1, cos(±π/2)=0)
    pitch = state.get('pitch', 0.0)
    upright_cos = math.cos(pitch)  # ranges [-1,1]; we shift to [0,1] later
    upright_cos = max(0.0, upright_cos)  # only reward when not overly tilted

    # Joint velocity penalty
    joint_vel_sum = 0.0
    for side in ('left', 'right'):
        for joint in ('hip', 'knee', 'wheel'):
            v = state.get(f'{side}_{joint}_velocity', 0.0)
            joint_vel_sum += v * v
    joint_vel_penalty = -0.00015 * joint_vel_sum

    # Torque penalty
    torque_sum = 0.0
    for side in ('left', 'right'):
        for joint in ('hip', 'knee', 'wheel'):
            t = state.get(f'{side}_{joint}_torque', 0.0)
            torque_sum += t * t
    torque_penalty = -0.00025 * torque_sum

    # Contact encouragement
    contact = state.get('contact', 0.0)
    contact_bonus = 0.012 * contact

    return {
        'speed_huber': speed_huber,
        'yaw_huber': yaw_huber,
        'upright_cos': upright_cos,
        'joint_vel_penalty': joint_vel_penalty,
        'torque_penalty': torque_penalty,
        'contact_bonus': contact_bonus,
    }

    Reward components:
    - forward_progress: displacement this step (ground_velocity * dt).
    - upright_cos2: Squared cosine of pitch (strong penalty for tilt).
    - yaw_rate_penalty: Penalize non-zero yaw rate unless commanded.
    - action_magnitude: Penalty on the magnitude of the action vector.
    - joint_energy: Penalty on squared joint velocities (effort).
    - contact_bonus: Bonus for wheel‑ground contact.
    """
    # Forward progress (metres per step)
    dt = state.get('dt', 0.01)
    ground_vel = state.get('ground_velocity', 0.0)
    forward_progress = ground_vel * dt  # reward for moving forward

    # Uprightness: use cos^2 to heavily penalize tilt
    pitch = state.get('pitch', 0.0)
    upright_cos2 = math.cos(pitch) * math.cos(pitch)  # [0,1]

    # Yaw rate: only reward if matches commanded yaw rate, otherwise penalize
    cmd_yaw = state.get('commanded_yaw_rate', 0.0)
    yaw_rate = state.get('yaw_rate', 0.0)
    yaw_err = cmd_yaw - yaw_rate
    yaw_rate_penalty = -0.4 * yaw_err * yaw_err  # quadratic penalty

    # Action magnitude penalty (encourages mild commands)
    ground_cmd = action.get('ground_velocity', 0.0)
    yaw_cmd = action.get('yaw_velocity', 0.0)
    action_mag = ground_cmd * ground_cmd + yaw_cmd * yaw_cmd
    action_penalty = -0.0008 * action_mag

    # Joint energy (velocity squared)
    joint_vel_sum = 0.0
    for side in ('left', 'right'):
        for joint in ('hip', 'knee', 'wheel'):
            v = state.get(f'{side}_{joint}_velocity', 0.0)
            joint_vel_sum += v * v
    joint_energy = -0.00012 * joint_vel_sum

    # Contact bonus
    contact = state.get('contact', 0.0)
    contact_bonus = 0.01 * contact

    return {
        'forward_progress': forward_progress,
        'upright_cos2': upright_cos2,
        'yaw_rate_penalty': yaw_rate_penalty,
        'action_penalty': action_penalty,
        'joint_energy': joint_energy,
        'contact_bonus': contact_bonus,
    }

    speed_exp = math.exp(-speed_err / 0.3)  # max 1 when error=0
    speed_reward = speed_exp - 1.0  # shift so zero error gives 0, negative otherwise

    # Yaw tracking similarly
    cmd_yaw = state.get('commanded_yaw_rate', 0.0)
    actual_yaw = state.get('yaw_rate', 0.0)
    yaw_err = abs(cmd_yaw - actual_yaw)
    yaw_exp = math.exp(-yaw_err / 0.5)
    yaw_reward = yaw_exp - 1.0

    # Pitch penalty (quadratic)

    pitch_pen = -0.8 * pitch * pitch  # stronger penalty for tilt

    # Jerk proxy: penalize angular rates of base
    pitch_rate = state.get('pitch_rate', 0.0)
    yaw_rate = state.get('yaw_rate', 0.0)
    jerk_proxy = -0.02 * (pitch_rate * pitch_rate + yaw_rate * yaw_rate)

    # Motor use: L1 norm of torques (encourages smaller torques)
    torque_l1 = 0.0
    for side in ('left', 'right'):
        for joint in ('hip', 'knee', 'wheel'):

            torque_l1 += abs(t)
    motor_use = -0.001 * torque_l1

    contact = state.get('contact', 0.0)
    contact_bonus = 0.008 * contact

        'speed_exp': speed_reward,
        'yaw_exp': yaw_reward,
        'pitch_pen': pitch_pen,
        'jerk_proxy': jerk_proxy,
        'motor_use': motor_use,
        'contact_bonus': contact_bonus,

import math

def compute_reward(state: dict, action: dict) -> dict:
    """
    Reward components:
    - speed_band: Reward if speed error lies within a tolerance band, else linear penalty.
    - yaw_band: Same for yaw rate.
    - upright_tanh: Smooth tanh-shaped reward for small pitch.
    - wheel_sync: Reward for left and right wheel velocities being similar (reduces slip).
    - joint_smooth: Penalty on joint acceleration approximated by change in velocity via dt? Not available → use joint velocity squared.
    - control_effort: Penalty on squared action values.
    """
    # Speed band reward
    cmd_speed = state.get('commanded_speed', 0.0)
    actual_speed = state.get('ground_velocity', 0.0)
    speed_err = cmd_speed - actual_speed
    band = 0.05  # m/s tolerance
    if abs(speed_err) <= band:
        speed_band = 0.1  # small reward for being inside band
    else:
        speed_band = -0.05 * (abs(speed_err) - band)  # linear penalty outside

    # Yaw band reward
    cmd_yaw = state.get('commanded_yaw_rate', 0.0)
    actual_yaw = state.get('yaw_rate', 0.0)
    yaw_err = cmd_yaw - actual_yaw
    yaw_band = 0.02
    if abs(yaw_err) <= yaw_band:
        yaw_band_reward = 0.05
    else:
        yaw_band_reward = -0.02 * (abs(yaw_err) - yaw_band)

    # Uprightness via tanh (smooth, saturates)
    pitch = state.get('pitch', 0.0)
    upright_tanh = math.tanh(1.0 / (abs(pitch) + 0.1))  # larger when pitch small
    # scale to roughly [0,1]
    upright_tanh = upright_tanh / math.tanh(1.0 / 0.1)

    # Wheel synchronization: reward when left and right wheel velocities match
    left_wheel_v = state.get('left_wheel_velocity', 0.0)
    right_wheel_v = state.get('right_wheel_velocity', 0.0)
    wheel_diff = left_wheel_v - right_wheel_v
    wheel_sync = -0.001 * (wheel_diff * wheel_diff)  # penalty for mismatch

    # Joint smoothness: penalize high joint velocities
    joint_vel_sq = 0.0
    for side in ('left', 'right'):
        for joint in ('hip', 'knee', 'wheel'):
            v = state.get(f'{side}_{joint}_velocity', 0.0)
            joint_vel_sq += v * v
    joint_smooth = -0.00008 * joint_vel_sq

    # Control effort: penalize squared action magnitudes
    ground_cmd = action.get('ground_velocity', 0.0)
    yaw_cmd = action.get('yaw_velocity', 0.0)
    control_effort = -0.0005 * (ground_cmd * ground_cmd + yaw_cmd * yaw_cmd)

    return {
        'speed_band': speed_band,
        'yaw_band': yaw_band_reward,
        'upright_tanh': upright_tanh,
        'wheel_sync': wheel_sync,
        'joint_smooth': joint_smooth,
        'control_effort': control_effort,
    }

import math

def compute_reward(state: dict, action: dict) -> dict:
    """
    Reward components:
    - progress_toward_goal: reward proportional to forward speed when commanded speed > 0.
    - stand_still_bonus: small bonus when commanded speed is zero and robot stays near zero speed.
    - upright_exp: Gaussian reward for pitch (max at zero).
    - yaw_align: Reward for matching yaw direction to commanded yaw rate using cosine similarity.
    - energy_cost: Penalty on sum of squared torques and joint velocities.
    - contact_maintain: Bonus for continuous contact.
    """
    cmd_speed = state.get('commanded_speed', 0.0)
    actual_speed = state.get('ground_velocity', 0.0)
    # When we want to move forward, reward actual forward speed; clip negative to avoid rewarding backward motion
    if cmd_speed > 0.0:
        progress = min(actual_speed, cmd_speed)  # reward up to commanded speed
        progress_toward = 0.6 * progress
    else:
        # Want to stay still: reward low speed
        progress_toward = 0.2 * (-abs(actual_speed))  # negative penalty for any movement

    # Stand‑still bonus when cmd_speed≈0
    stand_still = 0.0
    if abs(cmd_speed) < 0.02:
        stand_still = 0.05 * math.exp(-(actual_speed * actual_speed) / (2 * 0.02 * 0.02))

    # Upright Gaussian
    pitch = state.get('pitch', 0.0)
    upright_exp = math.exp(-(pitch * pitch) / (2 * 0.25 * 0.25))  # sigma=0.25 rad

    # Yaw alignment via cosine of angle difference (assuming small angles, cos ≈ 1 - 0.5*err^2)
    cmd_yaw = state.get('commanded_yaw_rate', 0.0)
    yaw_rate = state.get('yaw_rate', 0.0)
    yaw_diff = cmd_yaw - yaw_rate
    yaw_align = math.cos(yaw_diff)  # ranges [-1,1]; shift to [0,1]
    yaw_align = max(0.0, yaw_align)

    # Energy cost: torques + joint velocities
    energy = 0.0
    for side in ('left', 'right'):
        for joint in ('hip', 'knee', 'wheel'):
            t = state.get(f'{side}_{joint}_torque', 0.0)
            v = state.get(f'{side}_{joint}_velocity', 0.0)
            energy += 0.0001 * t * t + 0.00005 * v * v
    energy_cost = -energy

    # Contact maintenance
    contact = state.get('contact', 0.0)
    contact_maintain = 0.009 * contact

    return {
        'progress_toward': progress_toward,
        'stand_still_bonus': stand_still,
        'upright_exp': upright_exp,
        'yaw_align': yaw_align,
        'energy_cost': energy_cost,
        'contact_maintain': contact_maintain,
    }

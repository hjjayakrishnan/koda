import math

def compute_reward(state: dict, action: dict) -> dict:
    """
    Simple tracking‑plus‑stability reward.
    - speed_error: negative squared difference between commanded and actual forward speed.
    - upright: cosine of pitch (1 when upright, 0 when sideways, negative when inverted).
    - torque_penalty: sum of squared joint torques scaled down.
    - contact_bonus: small reward for maintaining wheel‑ground contact.
    """
    # speed tracking
    cmd_speed = state.get('commanded_speed', 0.0)
    actual_speed = state.get('ground_velocity', 0.0)
    speed_err = cmd_speed - actual_speed
    speed_error = -0.5 * speed_err * speed_err   # negative quadratic

    # uprightness (penalise lean)
    pitch = state.get('pitch', 0.0)
    upright = math.cos(pitch)  # 1 at pitch=0, decreases as |pitch| grows

    # torque effort
    torque_sum = 0.0
    for side in ('left', 'right'):
        for joint in ('hip', 'knee', 'wheel'):
            t = state.get(f'{side}_{joint}_torque', 0.0)
            torque_sum += t * t
    torque_penalty = -0.0002 * torque_sum

    # contact encouragement
    contact = state.get('contact', 0.0)
    contact_bonus = 0.01 * contact   # 0.01 when both wheels touch, 0 otherwise

    return {
        'speed_error': speed_error,
        'upright': upright,
        'torque_penalty': torque_penalty,
        'contact_bonus': contact_bonus,
    }

import math

def compute_reward(state: dict, action: dict) -> dict:
    """
    Progress‑and‑heading reward with smoothness regularisation.
    - forward_progress: ground_velocity * dt (meters moved).
    - heading_align: cosine of yaw to keep robot pointing forward.
    - pitch_penalty: squared pitch (keeps robot near vertical).
    - rate_smooth: penalises rapid changes in base rates (proxy for jerk).
    - action_reg: small penalty on magnitude of commanded velocities.
    """
    dt = state.get('dt', 0.01)
    ground_vel = state.get('ground_velocity', 0.0)
    forward_progress = ground_vel * dt

    yaw = state.get('yaw', 0.0)
    heading_align = math.cos(yaw)          # 1 when yaw≈0

    pitch = state.get('pitch', 0.0)
    pitch_penalty = -0.1 * pitch * pitch   # discourage lean

    # smoothness: penalise high pitch_rate and yaw_rate
    pr = state.get('pitch_rate', 0.0)
    yr = state.get('yaw_rate', 0.0)
    rate_smooth = -0.05 * (pr * pr + yr * yr)

    # action regularisation (encourage modest commands)
    gv_cmd = action.get('ground_velocity', 0.0)
    yv_cmd = action.get('yaw_velocity', 0.0)
    action_reg = -0.001 * (gv_cmd * gv_cmd + yv_cmd * yv_cmd)

    return {
        'forward_progress': forward_progress,
        'heading_align': heading_align,
        'pitch_penalty': pitch_penalty,
        'rate_smooth': rate_smooth,
        'action_reg': action_reg,
    }

import math

def compute_reward(state: dict, action: dict) -> dict:
    """
    Survival‑based reward with explicit fall detection.
    - alive_bonus: constant for staying upright (contact > 0).
    - fall_penalty: large negative when contact lost.
    - speed_match: reward for achieving commanded speed within a tolerance.
    - joint_limit_penalty: grows as any joint approaches ±π/2 rad (~90°).
    """
    # contact determines if robot is still standing
    contact = state.get('contact', 0.0)
    alive_bonus = 0.02 * contact          # 0.02 when touching ground, else 0
    fall_penalty = -1.0 * (1.0 - contact) # -1 if no contact

    # speed matching: reward inside a 0.05 m/s band, linearly decreasing outside
    cmd = state.get('commanded_speed', 0.0)
    vel = state.get('ground_velocity', 0.0)
    err = abs(cmd - vel)
    if err <= 0.05:
        speed_match = 0.03                # full bonus
    else:
        speed_match = max(0.0, 0.03 - 0.1 * (err - 0.05))

    # joint limit avoidance (soft barrier)
    limit_pen = 0.0
    for side in ('left', 'right'):
        for joint in ('hip', 'knee', 'wheel'):
            ang = abs(state.get(f'{side}_{joint}_position', 0.0))
            # treat π/2 ≈ 1.57 as hard limit; use quadratic penalty beyond 1.2 rad
            if ang > 1.2:
                excess = ang - 1.2
                limit_pen += excess * excess
    joint_limit_penalty = -0.02 * limit_pen

    return {
        'alive_bonus': alive_bonus,
        'fall_penalty': fall_penalty,
        'speed_match': speed_match,
        'joint_limit_penalty': joint_limit_penalty,
    }

import math

def compute_reward(state: dict, action: dict) -> dict:
    """
    Energy‑shaped reward: maximize forward work while minimizing metabolic cost.
    - mechanical_power: forward force * velocity approximated as wheel_torque * wheel_velocity.
    - metabolic_cost: sum of squared torques (effort proxy).
    - orientation_cost: penalises pitch and yaw rates squared.
    - progress_bonus: small reward for moving forward when power > 0.
    """
    # mechanical power approximation (only wheel contributions)
    power = 0.0
    for side in ('left', 'right'):
        t_w = state.get(f'{side}_wheel_torque', 0.0)
        v_w = state.get(f'{side}_wheel_velocity', 0.0)
        power += t_w * v_w   # Nm * rad/s ≈ Watts

    # metabolic cost (effort)
    cost = 0.0
    for side in ('left', 'right'):
        for joint in ('hip', 'knee', 'wheel'):
            tj = state.get(f'{side}_{joint}_torque', 0.0)
            cost += tj * tj
    metabolic = -0.0004 * cost

    # orientation cost (kinetic energy proxy)
    pr = state.get('pitch_rate', 0.0)
    yr = state.get('yaw_rate', 0.0)
    orient = -0.08 * (pr * pr + yr * yr)

    # progress bonus only when producing positive forward power
    progress_bonus = 0.005 * max(power, 0.0)

    return {
        'mechanical_power': power,
        'metabolic_cost': metabolic,
        'orientation_cost': orient,
        'progress_bonus': progress_bonus,
    }

import math

def compute_reward(state: dict, action: dict) -> dict:
    """
    Multi‑objective reward using hyperbolic tangent to bound terms.
    - speed_tanh: tanh of scaled speed error (reward when close to command).
    - pitch_tanh: negative tanh of absolute pitch (penalises lean).
    - yaw_tanh: negative tanh of absolute yaw (keeps heading straight).
    - effort_tanh: negative tanh of normalized total torque.
    - step_bonus: small constant for taking a step (encourages longevity).
    """
    cmd = state.get('commanded_speed', 0.0)
    vel = state.get('ground_velocity', 0.0)
    speed_err = cmd - vel
    speed_tanh = math.tanh(2.0 * speed_err)   # range [-1,1]; positive when slower than cmd

    pitch = state.get('pitch', 0.0)
    pitch_tanh = -math.tanh(abs(pitch))       # always ≤0, zero when upright

    yaw = state.get('yaw', 0.0)
    yaw_tanh = -math.tanh(abs(yaw))           # penalise heading drift

    # total torque normalisation (approx max 10 Nm per joint → 60 Nm)
    tot_torque = 0.0
    for side in ('left', 'right'):
        for joint in ('hip', 'knee', 'wheel'):
            tot_torque += abs(state.get(f'{side}_{joint}_torque', 0.0))
    effort_tanh = -math.tanh(tot_torque / 30.0)   # scale so typical values give modest penalty

    step_bonus = 0.001

    return {
        'speed_tanh': speed_tanh,
        'pitch_tanh': pitch_tanh,
        'yaw_tanh': yaw_tanh,
        'effort_tanh': effort_tanh,
        'step_bonus': step_bonus,
    }

import math

def compute_reward(state: dict, action: dict) -> dict:
    """
    Sparse‑style reward: give a bonus only when the robot is both upright enough
    and moving within a tolerance of the commanded speed; otherwise a small
    negative cost for actuation.
    - upright_reward: +0.04 if |pitch| < 0.2 rad (~11°) and contact == 1.
    - speed_reward: +0.03 if |cmd - vel| < 0.07 m/s.
    - actuation_cost: -0.0002 * sum of squared joint torques.
    - fallback_penalty: -0.001 applied every step (encourages achieving bonuses).
    """
    pitch = state.get('pitch', 0.0)
    contact = state.get('contact', 0.0)
    upright_ok = (abs(pitch) < 0.2) and (contact > 0.5)
    upright_reward = 0.04 if upright_ok else 0.0

    cmd = state.get('commanded_speed', 0.0)
    vel = state.get('ground_velocity', 0.0)
    speed_ok = abs(cmd - vel) < 0.07
    speed_reward = 0.03 if speed_ok else 0.0

    torque_sq = 0.0
    for side in ('left', 'right'):
        for joint in ('hip', 'knee', 'wheel'):
            t = state.get(f'{side}_{joint}_torque', 0.0)
            torque_sq += t * t
    actuation_cost = -0.0002 * torque_sq

    fallback_penalty = -0.001

    return {
        'upright_reward': upright_reward,
        'speed_reward': speed_reward,
        'actuation_cost': actuation_cost,
        'fallback_penalty': fallback_penalty,
    }

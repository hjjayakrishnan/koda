

import math

def compute_reward(state: dict, action: dict) -> dict:
    """Reward components:
    - speed_error: negative quadratic penalty for forward speed tracking.
    - yaw_error: negative quadratic penalty for yaw rate tracking.
    - upright_exp: exponential reward for staying near upright (max 1 at zero pitch).
    - power_cost: penalizes instantaneous mechanical power (|torque*velocity|) summed over joints.
    - contact_reward: small bonus when wheels are in contact with ground.
    - joint_symmetry: penalizes asymmetry between left and right hip/knee positions."""
    # speed tracking
    cmd_speed = state.get('commanded_speed', 0.0)
    act_speed = state.get('ground_velocity', 0.0)
    sp_err = cmd_speed - act_speed
    speed_error = -0.6 * sp_err * sp_err

    # yaw tracking
    cmd_yaw = state.get('commanded_yaw_rate', 0.0)
    act_yaw = state.get('yaw_rate', 0.0)
    y_err = cmd_yaw - act_yaw
    yaw_error = -0.4 * y_err * y_err

    # uprightness via exponential (softer than Gaussian near zero)
    pitch = state.get('pitch', 0.0)
    upright_exp = math.exp(-abs(pitch) / 0.15)  # decays to ~0.5 at 0.15 rad (~8.6°)

    # power consumption (torque * angular velocity) penalty
    power_sum = 0.0
    for side in ('left', 'right'):
        for joint in ('hip', 'knee', 'wheel'):
            t = state.get(f'{side}_{joint}_torque', 0.0)
            v = state.get(f'{side}_{joint}_velocity', 0.0)
            power_sum += abs(t * v)
    power_cost = -0.0005 * power_sum

    # contact encouragement
    contact = state.get('contact', 0.0)
    contact_reward = 0.02 * contact

    # symmetry reward/punishment: aim for mirrored hip/knee angles
    sym_err = 0.0
    for joint in ('hip', 'knee'):
        lp = state.get(f'left_{joint}_position', 0.0)
        rp = state.get(f'right_{joint}_position', 0.0)
        sym_err += (lp + rp) ** 2  # penalize when sum != 0 (mirrored about zero)
    joint_symmetry = -0.001 * sym_err

    return {
        'speed_error': speed_error,
        'yaw_error': yaw_error,
        'upright_exp': upright_exp,
        'power_cost': power_cost,
        'contact_reward': contact_reward,
        'joint_symmetry': joint_symmetry,
    }

        stability_pitch = -5.0

        stability_pick = 0.0
        stability_pitch = 0.0  # no penalty within threshold
    # correct variable name
    stability_pitch = -5.0 if abs(pitch) > pitch_thresh else 0.0

    cmd_yaw = state.get('commanded_yaw_rate', 0.0)
    act_yaw = state.get('yaw_rate', 0.0)

    norm_act = math.hypot(act_yaw, 1e-8)
    yaw_cos = (cmd_yaw * act_yaw) / (norm_cmd * norm_act) if norm_cmd * norm_act != 0 else 1.0
    yaw_align = 0.3 * yaw_cos

    # jerk proxy: penalize rapid torque changes (use wheel torques as proxy)
    jerk = 0.0
    for side in ('left', 'right'):
        t = state.get(f'{side}_wheel_torque', 0.0)
        jerk += t * t
    jerk_penalty = -0.0004 * jerk

    # energy cost (torque squared)
    energy = 0.0
    for side in ('left', 'right'):
        for joint in ('hip', 'knee', 'wheel'):
            t = state.get(f'{side}_{joint}_torque', 0.0)
            energy += t * t
    energy_cost = -0.00015 * energy

    # idle bonus: when no movement requested, reward staying put
    idle = 0.0

        idle = 0.02 * (1.0 - min(abs(act_speed), 0.5) / 0.5)  # higher bonus if actually slow

        idle = 0.0

    return {
        'progress_reward': progress_reward,
        'stability_pitch': stability_pitch,

                joint_avoid += -1.0 / (dist * dist + 1e-3)  # blows up near limit

                joint_avoid += -5.0  # already past limit (should not happen)

    smooth = 0.0

def compute_reward(state: dict, action: dict) -> dict:
    """Reward based on task-oriented potentials:
    - speed_track_linear: linear reward for moving at commanded speed (negative if opposite).
    - yaw_rate_match: reward proportional to product of commanded and actual yaw rate.
    - upright_cosine: cosine‑based reward for pitch (max 1 at zero, 0 at ±π/2).
    - minimal_motion: penalizes unnecessary joint motion when no command.
    - torque_rate: penalizes rate of change of torque approximated by torque itself.
    - energy_efficiency: reward for low power consumption when moving."""
    # speed track – linear, encourages correct direction
    cmd_sp = state.get('commanded_speed', 0.0)
    act_sp = state.get('ground_velocity', 0.0)
    speed_track_lin = 1.0 * cmd_sp * act_sp  # positive when signs match, zero if no command

    # yaw rate match
    cmd_yaw = state.get('commanded_yaw_rate', 0.0)
    act_yaw = state.get('yaw_rate', 0.0)
    yaw_rate_match = 0.5 * cmd_yaw * act_yaw  # positive when aligned

    # upright cosine (bounded)
    pitch = state.get('pitch', 0.0)
    upright_cos = math.cos(pitch)  # cos(0)=1, cos(±π/2)=0, negative beyond
    # clip to [0,1] to avoid rewarding upside down
    if upright_cos < 0.0:
        upright_cos = 0.0

    # minimal motion: when no command, penalize any joint movement
    no_cmd = (abs(cmd_sp) < 1e-3) and (abs(cmd_yaw) < 1e-3)
    joint_move = 0.0
    if no_cmd:
        for side in ('left', 'right'):
            for joint in ('hip', 'knee', 'wheel'):
                v = state.get(f'{side}_{joint}_velocity', 0.0)
                joint_move += abs(v)
        minimal_motion = -0.01 * joint_move
    else:
        minimal_motion = 0.0

    # torque rate penalty (use torque as proxy for its derivative)
    torque_rate = 0.0
    for side in ('left', 'right'):
        for joint in ('hip', 'knee', 'wheel'):
            t = state.get(f'{side}_{joint}_torque', 0.0)
            torque_rate += t * t
    torque_rate = -0.0003 * torque_rate

    # energy efficiency: reward low power when actually moving forward
    power = 0.0
    for side in ('left', 'right'):
        for joint in ('hip', 'knee', 'wheel'):
            t = state.get(f'{side}_{joint}_torque', 0.0)
            v = state.get(f'{side}_{joint}_velocity', 0.0)
            power += abs(t * v)
    # only give benefit when forward speed positive
    eff_bonus = 0.0
    if act_sp > 0.0:
        eff_bonus = -0.0004 * power  # negative penalty, smaller power -> less negative => higher reward
    else:
        eff_bonus = 0.0

    return {
        'speed_track_lin': speed_track_lin,
        'yaw_rate_match': yaw_rate_match,
        'upright_cos': upright_cos,
        'minimal_motion': minimal_motion,
        'torque_rate': torque_rate,
        'eff_bonus': eff_bonus,
    }
